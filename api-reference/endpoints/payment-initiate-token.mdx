---
title: "Initiate Payment Token"
api: "POST /Gateway/v{version}/Payment/initiate-token"
description: "Generate a payment token for custom checkout implementation"
---

## Overview

This endpoint creates a payment intent and returns a token that can be used to implement your own custom checkout flow. The token can be used to charge mobile wallets or process payouts.

## Path Parameters

<ParamField path="version" type="string" required>
  API version (e.g., "1")
</ParamField>

## Request Body

<ParamField body="amount" type="number" required>
  Payment amount (must be greater than 0.01)
</ParamField>

<ParamField body="currency" type="string">
  Payment currency (e.g., "PKR", "USD")
</ParamField>

<ParamField body="description" type="string" required>
  Payment description (1-200 alphanumeric characters and spaces only)
</ParamField>

<ParamField body="customerRef" type="object">
  Customer information
  <Expandable title="properties">
    <ParamField body="name" type="string" required>
      Customer full name
    </ParamField>
    <ParamField body="email" type="string" required>
      Customer email address
    </ParamField>
    <ParamField body="cnic" type="string" required>
      Customer CNIC number
    </ParamField>
    <ParamField body="phone" type="string" required>
      Customer phone number
    </ParamField>
    <ParamField body="shippingAddress" type="string">
      Shipping address
    </ParamField>
    <ParamField body="estimatedDays" type="string">
      Estimated delivery days
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="orderRef" type="object">
  Order information
  <Expandable title="properties">
    <ParamField body="orderRef" type="string" required>
      Order reference (1-20 alphanumeric characters)
    </ParamField>
    <ParamField body="discount" type="string">
      Discount amount or percentage
    </ParamField>
    <ParamField body="tax" type="string">
      Tax amount or percentage
    </ParamField>
  </Expandable>
</ParamField>

<ParamField body="successUrl" type="string">
  URL to redirect after successful payment
</ParamField>

<ParamField body="cancelUrl" type="string">
  URL to redirect after cancelled payment
</ParamField>

<ParamField body="intentType" type="string">
  Payment intent type. Allowed values: `checkout`, `deposit`, `withdrawal`, `refund`, `settlement`
</ParamField>

<ParamField body="metadataJson" type="string">
  Additional metadata in JSON format
</ParamField>

## Response

<ResponseField name="paymentIntentId" type="uuid">
  Unique identifier for the payment intent
</ResponseField>

<ResponseField name="token" type="string">
  Payment token to use for charging or payout
</ResponseField>

<ResponseField name="expiresAt" type="datetime">
  Token expiration timestamp
</ResponseField>
